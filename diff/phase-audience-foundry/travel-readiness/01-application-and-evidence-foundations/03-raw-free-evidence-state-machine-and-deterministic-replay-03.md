+    def test_success_finalization_and_artifact_creation_are_linearized(self):
+        body_hash = "c" * 64
+        anchor_completed_at = timezone.now() + timedelta(seconds=4)
+        first = self.successful_attempt(
+            self.source,
+            self.rights,
+            body_hash,
+            started_at=anchor_completed_at - timedelta(seconds=4),

+        pending = self.start_attempt(
+            self.source,
+            self.rights,
+            started_at=anchor_completed_at - timedelta(seconds=3),
+        )
+        barrier = threading.Barrier(2)
+        outcomes = []
+        unexpected = []
+        result_lock = threading.Lock()
+
+        def record(value):
+            with result_lock:
+                outcomes.append(value)
+
+        def create_artifact():
+            close_old_connections()
+            try:
+                barrier.wait()
+                with transaction.atomic():
+                    SourceArtifact.objects.create(
+                        source_id=self.source.pk,
+                        body_sha256=body_hash,
+                        byte_count=10,
+                        first_successful_attempt_id=first.pk,
+                    )
+                record("artifact_created")
+            except IntegrityError:
+                record("artifact_blocked")
+            except BaseException as exc:
+                with result_lock:
+                    unexpected.append(type(exc).__name__)
+            finally:
+                close_old_connections()
+
+        def finalize_success():
+            close_old_connections()
+            try:
+                barrier.wait()
+                with transaction.atomic():
+                    FetchAttempt.objects.filter(pk=pending.pk).update(
+                        result=FetchAttempt.Result.SUCCEEDED,
+                        completed_at=anchor_completed_at - timedelta(seconds=1),
+                        http_status=200,
+                        response_sha256=body_hash,
+                    )
+                record("success_finalized")
+            except IntegrityError:
+                record("success_blocked")
+            except BaseException as exc:
+                with result_lock:
+                    unexpected.append(type(exc).__name__)
+            finally:
+                close_old_connections()
+
+        workers = [threading.Thread(target=create_artifact), threading.Thread(target=finalize_success)]
+        for worker in workers:
+            worker.start()
+        for worker in workers:
+            worker.join(timeout=10)

+        self.assertFalse(any(worker.is_alive() for worker in workers))
+        self.assertEqual(unexpected, [])
+        self.assertIn(

+                {"artifact_created", "success_blocked"},

+            result=FetchAttempt.Result.SUCCEEDED,

+        artifact = SourceArtifact.objects.filter(

+                body_sha256=body_hash,
