+from django.db import IntegrityError, transaction
+from django.test import TestCase
+from django.utils import timezone
+
+from sources.models import SourceConfiguration
+
+
+class SourceConfigurationTests(TestCase):
+    def valid(self, **overrides):
+        values = {
+            "code": "MOFA_ENTRY_CSV",
+            "revision": "rights-v1",
+            "module": SourceConfiguration.Module.ENTRY,
+            "owner": "MOFA",
+            "official_locator": "https://example.invalid/source.csv",
+        }
+        values.update(overrides)
+        return SourceConfiguration.objects.create(**values)
+
+    def assert_rejected(self, **overrides):
+        with self.assertRaises(IntegrityError), transaction.atomic():
+            self.valid(**overrides)
+
+    def test_stable_code_and_utc_creation(self):
+        source = self.valid()
+        self.assertTrue(timezone.is_aware(source.created_at))
+        self.assert_rejected(revision="rights-v2")
+        with self.assertRaises(IntegrityError), transaction.atomic():
+            SourceConfiguration.objects.filter(pk=source.pk).update(code="MOFA_RENAMED")
+
+    def test_state_and_enabled_are_consistent(self):
+        self.assert_rejected(state=SourceConfiguration.State.ACTIVE)
+        self.assert_rejected(enabled=True)
+        self.assert_rejected(state=SourceConfiguration.State.ACTIVE, enabled=True)
+        source = self.valid()
+        with self.assertRaises(IntegrityError), transaction.atomic():
+            SourceConfiguration.objects.filter(pk=source.pk).update(
+                state=SourceConfiguration.State.RIGHTS_APPROVED
+            )
+
+    def test_contract_change_requires_new_draft_revision(self):
+        source = self.valid()
+        with self.assertRaises(IntegrityError), transaction.atomic():
+            SourceConfiguration.objects.filter(pk=source.pk).update(owner="Changed owner")
+        SourceConfiguration.objects.filter(pk=source.pk).update(
+            owner="Changed owner", revision="rights-v2"
+        )
+        source.refresh_from_db()
+        self.assertEqual(source.revision, "rights-v2")
+        self.assertEqual(source.state, SourceConfiguration.State.DRAFT)
+
+    def test_transport_settings_are_bounded(self):
+        self.assert_rejected(connect_timeout_seconds=0)
+        self.assert_rejected(read_timeout_seconds=61)
+        self.assert_rejected(max_retries=4)
+        self.assert_rejected(official_locator="http://example.invalid/source.csv")
+        self.assert_rejected(code="lowercase")
+        self.assert_rejected(revision="../rights")
+        self.assert_rejected(owner="")
+        self.assert_rejected(module="UNKNOWN")
+        self.assert_rejected(secret_reference_name="literal-value")
+
+    def test_only_a_secret_reference_name_is_present(self):
+        names = {field.name for field in SourceConfiguration._meta.fields}
+        self.assertIn("secret_reference_name", names)
+        self.assertTrue({"password", "credential", "api_key", "secret_value"}.isdisjoint(names))
